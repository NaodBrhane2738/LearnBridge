Integration Snippet for ALX Savanna Developers

HTML / Vanilla JS 
<!-- 1. Include LearnBridge Web Components bundle in Savanna's <head> -->
<script type="module" src="https://savanna.alx.et/cdn/learnbridge.js"></script>

<!-- 2. Place components anywhere in the lesson template -->
<learnbridge-player 
  id="player"
  src="/media/lessons/04-accessibility.mp4"
  captions-en="/media/captions/04-en.vtt"
  captions-am="/media/captions/04-am.vtt"
  audio-desc="/media/audio/04-desc.mp3">
</learnbridge-player>

<learnbridge-transcript 
  id="transcript"
  src="/media/transcripts/04.json">
</learnbridge-transcript>

<!-- 3. Wire host events with 3 lines of JavaScript -->
<script>
  const player = document.getElementById('player');
  const transcript = document.getElementById('transcript');

  // Player updates transcript time
  player.addEventListener('lb-timeupdate', (e) => {
    transcript.setTime(e.detail.currentTime);
  });

  // Transcript click seeks player
  transcript.addEventListener('lb-seek-request', (e) => {
    player.seek(e.detail.time);
  });
</script>
